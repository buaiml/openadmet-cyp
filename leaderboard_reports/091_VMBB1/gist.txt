# 1. Load Datasets
ds = load_dataset("openadmet/cyp-challenge-train-test")
train_df = ds["train"].to_pandas()
test_df = ds["test"].to_pandas()

# 2. Featurizer: ECFP4 Fingerprints + Key RDKit Descriptors
def generate_features(smiles_list):
    fps, desc_list = [], []
    for s in smiles_list:
        mol = Chem.MolFromSmiles(s)
        if mol is None:
            fps.append(np.zeros(2048))
            desc_list.append([0]*4)
            continue
        # ECFP4
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, radius=2, nBits=2048)
        fps.append(np.array(fp))
        # Basic Descriptors
        desc_list.append([
            Descriptors.MolWt(mol),
            Descriptors.MolLogP(mol),
            Descriptors.TPSA(mol),
            Descriptors.NumRotatableBonds(mol)
        ])
    return np.hstack([np.array(fps), np.array(desc_list)])

X_train = generate_features(train_df["SMILES"])
X_test = generate_features(test_df["SMILES"])

# 3. Scaffold-Based Grouping for Cross-Validation
def get_scaffold(smiles):
    mol = Chem.MolFromSmiles(smiles)
    if mol is None: return ""
    return MurckoScaffold.MurckoScaffoldSmiles(mol=mol)

scaffolds = train_df["SMILES"].apply(get_scaffold)
_, groups = np.unique(scaffolds, return_inverse=True)

# 4. Train Multi-Task Multi-Output Ensemble (Track 1 Regression Example)
targets = [
    'CYP1A2_pIC50_direct_inhibition', 
    'CYP2C9_pIC50_direct_inhibition',
    'CYP2D6_pIC50_direct_inhibition', 
    'CYP3A4_pIC50_direct_inhibition'
]

test_preds = np.zeros((len(test_df), len(targets)))
gkf = GroupKFold(n_splits=5)

for i, target in enumerate(targets):
    y = train_df[target].values
    valid_mask = ~np.isnan(y)  # handle missing labels if any
    
    oof_preds = np.zeros(len(train_df))
    
    for train_idx, val_idx in gkf.split(X_train[valid_mask], y[valid_mask], groups[valid_mask]):
        X_tr, y_tr = X_train[valid_mask][train_idx], y[valid_mask][train_idx]
        X_va, y_va = X_train[valid_mask][val_idx], y[valid_mask][val_idx]
        
        model = lgb.LGBMRegressor(
            n_estimators=500,
            learning_rate=0.03,
            num_leaves=31,
            random_state=42
        )
        model.fit(
            X_tr, y_tr,
            eval_set=[(X_va, y_va)],
            callbacks=[lgb.early_stopping(50, verbose=False)]
        )
        
        test_preds[:, i] += model.predict(X_test) / 5.0

# 5. Format Track 1 Submission File
submission_df = pd.DataFrame({
    'Molecule_Name': test_df['Molecule_Name'],
    'SMILES': test_df['SMILES'],
    'CYP1A2_pIC50_direct_inhibition': test_preds[:, 0],
    'CYP2C9_pIC50_direct_inhibition': test_preds[:, 1],
    'CYP2D6_pIC50_direct_inhibition': test_preds[:, 2],
    'CYP3A4_pIC50_direct_inhibition': test_preds[:, 3],
})
submission_df.to_csv("submission_track1.csv", index=False)
